topic通信死锁

在进行topic通信时，没有报错，但没有任何反应，发现是发生了死锁

2026-04-27
2 minute read

问题描述：

在使用arm_base和arm_cmd进行通信的时候，发现编译没有报错，并且也成功启动，但是两个话题之间没有进行通信，在经过排查之后发现，原来是两个节点之间发生死锁。

原因：

通过查看两个节点的构造函数

ArmCmd::ArmCmd()
    : rclcpp::Node("arm_cmd"),
      cur_x_(kDefaultX),
      cur_y_(kDefaultY),
      cur_z_(kDefaultZ),
      cmd_sent_(false) {
  this->declare_parameter<float>("target_x", kDefaultTargetX);
  this->declare_parameter<float>("target_y", kDefaultTargetY);
  this->declare_parameter<float>("target_z", kDefaultTargetZ);
  this->get_parameter("target_x", target_x_);
  this->get_parameter("target_y", target_y_);
  this->get_parameter("target_z", target_z_);

  arm_cmd_pub_ = create_publisher<arm_msg::msg::Arm>("arm_cmd", 10);

  arm_status_sub_ = create_subscription<arm_msg::msg::ArmStatus>(
      "arm_status", rclcpp::QoS(1).transient_local(),
      [this](const arm_msg::msg::ArmStatus::SharedPtr msg) {
        StatusCallback(msg);
      });

  RCLCPP_INFO(this->get_logger(), "arm_cmd 已启动，目标位置: (%.2f, %.2f, %.2f)",
              target_x_, target_y_, target_z_);
}

ArmBase::ArmBase() : rclcpp::Node("arm_base"), serial_baud_rate_(115200) {
  memset(&send_data_, 0, sizeof(send_data_));
  current_pos_.x = kDefaultPosX;
  current_pos_.y = kDefaultPosY;
  current_pos_.z = kDefaultPosZ;

  this->declare_parameter<std::string>("usart_port_name", "/dev/ttyUSB0");
  this->declare_parameter<int>("serial_baud_rate", 9600);
  this->get_parameter("usart_port_name", usart_port_name_);
  this->get_parameter("serial_baud_rate", serial_baud_rate_);

  arm_cmd_sub_ = create_subscription<arm_msg::msg::Arm>(
      "arm_cmd", 10,
      [this](const arm_msg::msg::Arm::SharedPtr cmd) { ArmCmdCallback(cmd); });

  // transient_local：晚订阅的节点也能收到最后一条状态，避免启动时序竞争
  arm_status_pub_ = create_publisher<arm_msg::msg::ArmStatus>(
      "arm_status", rclcpp::QoS(1).transient_local());

  try {
    arm_serial_.setPort(usart_port_name_);
    arm_serial_.setBaudrate(serial_baud_rate_);
    serial::Timeout t = serial::Timeout::simpleTimeout(100);
    arm_serial_.setTimeout(t);
    arm_serial_.open();
  } catch (serial::IOException &e) {
    RCLCPP_ERROR(this->get_logger(), "无法打开串口 %s: %s",
                 usart_port_name_.c_str(), e.what());
  }

  if (arm_serial_.isOpen())
    RCLCPP_INFO(this->get_logger(), "串口已打开: %s @ %d",
                usart_port_name_.c_str(), serial_baud_rate_);
}

发现arm_cmd一直在等待接收来自arm_base发送的消息ArmStatus才进入回调，而arm_base构造函数里没有第一时间发送ArmStatus，导致死锁。

解决：

在arm_base的构造函数里先主动发送一次Armstatus

芯片信息

  • 芯片：STM32F103C8T6（Blue Pill）
  • 主频：72 MHz（HSE 8 MHz × PLL × 9）
  • 工具链：STM32CubeMX + HAL + Keil MDK-ARM

void SystemClock_Config(void) {
    RCC_OscInitTypeDef RCC_OscInitStruct = {0};
    RCC_ClkInitTypeDef RCC_ClkInitStruct = {0};

    // 使能 HSE，PLL 倍频 ×9
    RCC_OscInitStruct.OscillatorType = RCC_OSCILLATORTYPE_HSE;
    RCC_OscInitStruct.HSEState       = RCC_HSE_ON;
    RCC_OscInitStruct.HSEPredivValue = RCC_HSE_PREDIV_DIV1;
    RCC_OscInitStruct.PLL.PLLState   = RCC_PLL_ON;
    RCC_OscInitStruct.PLL.PLLSource  = RCC_PLLSOURCE_HSE;
    RCC_OscInitStruct.PLL.PLLMUL     = RCC_PLL_MUL9;  // 8 MHz × 9 = 72 MHz
    HAL_RCC_OscConfig(&RCC_OscInitStruct);

    // 配置总线分频
    RCC_ClkInitStruct.ClockType      = RCC_CLOCKTYPE_HCLK | RCC_CLOCKTYPE_SYSCLK
                                     | RCC_CLOCKTYPE_PCLK1 | RCC_CLOCKTYPE_PCLK2;
    RCC_ClkInitStruct.SYSCLKSource   = RCC_SYSCLKSOURCE_PLLCLK;
    RCC_ClkInitStruct.AHBCLKDivider  = RCC_SYSCLK_DIV1;   // HCLK  = 72 MHz
    RCC_ClkInitStruct.APB1CLKDivider = RCC_HCLK_DIV2;     // APB1  = 36 MHz（最大 36 MHz）
    RCC_ClkInitStruct.APB2CLKDivider = RCC_HCLK_DIV1;     // APB2  = 72 MHz
    HAL_RCC_ClockConfig(&RCC_ClkInitStruct, FLASH_LATENCY_2);
}
时钟域 频率 说明
HSE 8 MHz 外部晶振
SYSCLK 72 MHz PLL 输出，系统主频
HCLK (AHB) 72 MHz CPU、DMA、Flash
PCLK1 (APB1) 36 MHz TIM2-7、USART2/3、SPI2、I2C
PCLK2 (APB2) 72 MHz USART1、SPI1、TIM1、ADC、GPIO

FLASH_LATENCY_2 是 72 MHz 时 Flash 访问所需的等待周期数，必须与主频匹配，否则程序跑飞。

void MX_GPIO_Init(void) {
    GPIO_InitTypeDef GPIO_InitStruct = {0};

    // 使能端口时钟
    __HAL_RCC_GPIOA_CLK_ENABLE();
    __HAL_RCC_GPIOD_CLK_ENABLE();  // HSE 晶振引脚所在端口

    // 初始化输出电平为低
    HAL_GPIO_WritePin(LED_GPIO_Port, LED_Pin, GPIO_PIN_RESET);

    // LED：推挽输出
    GPIO_InitStruct.Pin   = LED_Pin;
    GPIO_InitStruct.Mode  = GPIO_MODE_OUTPUT_PP;
    GPIO_InitStruct.Pull  = GPIO_NOPULL;
    GPIO_InitStruct.Speed = GPIO_SPEED_FREQ_LOW;
    HAL_GPIO_Init(LED_GPIO_Port, &GPIO_InitStruct);

    // KEY：浮空输入（轮询）
    GPIO_InitStruct.Pin  = KEY_Pin;
    GPIO_InitStruct.Mode = GPIO_MODE_INPUT;
    GPIO_InitStruct.Pull = GPIO_NOPULL;
    HAL_GPIO_Init(KEY_GPIO_Port, &GPIO_InitStruct);
}

GPIO 模式说明

模式宏 说明
GPIO_MODE_OUTPUT_PP 推挽输出，可驱动高低电平
GPIO_MODE_OUTPUT_OD 开漏输出，需外部上拉才能输出高电平
GPIO_MODE_INPUT 浮空输入，引脚电平由外部决定
GPIO_MODE_AF_PP 复用推挽，引脚由外设（USART/SPI 等）控制
GPIO_MODE_IT_FALLING 下降沿触发外部中断

GPIO 时钟默认关闭，使用前必须通过 __HAL_RCC_GPIOx_CLK_ENABLE() 使能，否则写引脚无效。

void MX_GPIO_Init(void) {
    GPIO_InitTypeDef GPIO_InitStruct = {0};

    __HAL_RCC_GPIOA_CLK_ENABLE();
    __HAL_RCC_GPIOB_CLK_ENABLE();
    __HAL_RCC_GPIOD_CLK_ENABLE();

    HAL_GPIO_WritePin(LED_GPIO_Port, LED_Pin, GPIO_PIN_RESET);

    // LED：推挽输出
    GPIO_InitStruct.Pin   = LED_Pin;
    GPIO_InitStruct.Mode  = GPIO_MODE_OUTPUT_PP;
    GPIO_InitStruct.Pull  = GPIO_NOPULL;
    GPIO_InitStruct.Speed = GPIO_SPEED_FREQ_LOW;
    HAL_GPIO_Init(LED_GPIO_Port, &GPIO_InitStruct);

    // KEY：下降沿中断 + 内部上拉
    GPIO_InitStruct.Pin  = KEY_Pin;
    GPIO_InitStruct.Mode = GPIO_MODE_IT_FALLING;  // 下降沿触发
    GPIO_InitStruct.Pull = GPIO_PULLUP;            // 内部上拉，按下为低
    HAL_GPIO_Init(KEY_GPIO_Port, &GPIO_InitStruct);

    // 使能 EXTI15_10 中断（PB11 在 EXTI 线 11，属于 EXTI15_10 组）
    HAL_NVIC_SetPriority(EXTI15_10_IRQn, 0, 0);
    HAL_NVIC_EnableIRQ(EXTI15_10_IRQn);
}

EXTI 线与引脚的对应关系

STM32F1 的 EXTI 线与引脚编号一一对应，但每条线只能连接一个端口：

EXTI 线 可选引脚 IRQ Handler
EXTI0 PA0/PB0/PC0/… EXTI0_IRQHandler
EXTI1~4 Px1~Px4 EXTI1_IRQHandler 等
EXTI5~9 Px5~Px9 EXTI9_5_IRQHandler
EXTI10~15 Px10~Px15 EXTI15_10_IRQHandler

PB11 使用 EXTI 线 11，属于 EXTI15_10_IRQn 组，所以使能 EXTI15_10_IRQn。

触发方式

模式宏 触发条件
GPIO_MODE_IT_RISING 上升沿（低→高）
GPIO_MODE_IT_FALLING 下降沿（高→低）
GPIO_MODE_IT_RISING_FALLING 双边沿

按键接地、内部上拉时，按下产生下降沿，选 GPIO_MODE_IT_FALLING。

void MX_USART1_UART_Init(void) {
    huart1.Instance          = USART1;
    huart1.Init.BaudRate     = 115200;
    huart1.Init.WordLength   = UART_WORDLENGTH_8B;
    huart1.Init.StopBits     = UART_STOPBITS_1;
    huart1.Init.Parity       = UART_PARITY_NONE;
    huart1.Init.Mode         = UART_MODE_TX_RX;
    huart1.Init.HwFlowCtl    = UART_HWCONTROL_NONE;
    huart1.Init.OverSampling = UART_OVERSAMPLING_16;
    HAL_UART_Init(&huart1);  // 内部调用 HAL_UART_MspInit
}
参数 值 说明
BaudRate 115200 波特率，串口工具需一致
WordLength 8B 每帧 8 位数据
StopBits 1 1 位停止位
Parity NONE 无校验位
Mode TX_RX 同时使能发送和接收
HwFlowCtl NONE 不使用硬件流控
OverSampling 16 16 倍过采样，抗噪声能力强

void HAL_UART_MspInit(UART_HandleTypeDef *uartHandle) {
    GPIO_InitTypeDef GPIO_InitStruct = {0};
    if (uartHandle->Instance == USART1) {
        // 使能外设时钟
        __HAL_RCC_USART1_CLK_ENABLE();
        __HAL_RCC_GPIOA_CLK_ENABLE();

        // PA9 (TX)：复用推挽输出，由 USART1 外设驱动
        GPIO_InitStruct.Pin   = GPIO_PIN_9;
        GPIO_InitStruct.Mode  = GPIO_MODE_AF_PP;
        GPIO_InitStruct.Speed = GPIO_SPEED_FREQ_HIGH;
        HAL_GPIO_Init(GPIOA, &GPIO_InitStruct);

        // PA10 (RX)：浮空输入，由对端设备驱动
        GPIO_InitStruct.Pin  = GPIO_PIN_10;
        GPIO_InitStruct.Mode = GPIO_MODE_INPUT;
        GPIO_InitStruct.Pull = GPIO_NOPULL;
        HAL_GPIO_Init(GPIOA, &GPIO_InitStruct);

        // 配置 USART1 中断
        HAL_NVIC_SetPriority(USART1_IRQn, 0, 0);
        HAL_NVIC_EnableIRQ(USART1_IRQn);
    }
}
引脚 方向 GPIO 模式 说明
PA9 TX（输出） GPIO_MODE_AF_PP 复用推挽，USART1 外设控制
PA10 RX（输入） GPIO_MODE_INPUT + GPIO_NOPULL 浮空输入，外部信号驱动

TX 必须配置为复用推挽（AF_PP），让 USART1 外设直接控制引脚电平。RX 配置为浮空输入，由对端设备驱动。

接线 

1
2
3

STM32 PA9  (TX)  →  USB 转串口模块 RXD
STM32 PA10 (RX)  →  USB 转串口模块 TXD
STM32 GND        →  USB 转串口模块 GND

TX 与 RXD 交叉连接，同名相接（TX→TXD）会导致 PC 端收不到数据。

初始化调用顺序 

1
2
3
4

HAL_Init();             // 初始化 HAL，配置 SysTick（1 ms）
SystemClock_Config();   // 配置 PLL，主频 72 MHz
MX_GPIO_Init();         // 使能 GPIO 时钟，配置引脚
MX_USART1_UART_Init();  // 初始化 USART1（内部调用 MspInit）

MX_GPIO_Init 必须在 MX_USART1_UART_Init 之前，因为 USART 初始化依赖 GPIOA 时钟已使能。

Ros2学习笔记

(一)Ros2 架构

ros2 主要可分为三层：操作系统层、中间件层、应用层

  • 操作系统层：使用的操作系统
  • 中间件层：
    • DDS层 (Data Distribution Service)：这就相当于物流系统的底层网络（公路、飞机），是ROS2实现通信的核心协议。它负责把数据按照你要求的服务质量（QoS，比如要速度快还是要绝不丢包）从一个节点安全送达另一个节点。
    • 抽象DDS层-RMW (ROS Middleware)：市面上有很多家不同的物流公司（比如FastRTPS, CycloneDDS等不同的DDS实现）。RMW 就像是一个中间平台，它把不同厂家的差异抹平了。有了它，当你写代码时，你可以随时更换底层的DDS提供商，而不需要修改你的一行应用代码。
    • RCL客户端层 (ROS Client Library)：这就好比物流系统的营业员，是直接面向我们程序员的接口。它把底层的复杂操作封装成了简单的函数。我们日常写代码就是用 RCL 提供的各种语言的API，比如 C++ 用 rclcpp，Python 用 rclpy。
  • 应用层：就是我们写的代码以及开发的工具这些

(二)Ros2基础知识

1. Ros2节点

1.1. 节点是什么？

ROS2中每一个节点就像你身体里的每一个器官，只负责某个功能，但和其他器官共同合作(比如一个节点负责控制车轮转动，一个节点负责从激光雷达获取数据、一个节点负责处理激光雷达的数据、一个节点负责定位等等)

1.2. 节点之间如何通信？

ros2一共有四种通信方式：

1. topics
2. services
3. Action
4. parameters

2.3. colcon使用方法

colcon build 必须是在工作空间目录下

  • 常用参数
    • 构建某个功能包 --packages-select
    • 构建某个工能包以及其依赖的包 --packages-up-to
    • 构建某个功能包以及依赖它的包 --packages-above
    • 某个包编译失败继续编译 --continue-on-error
    • 安装时软连接(Python开发时，修改代码不用重编译) --symlink-install
    • 指定线程数 -jN
  • 构建完成后一定要 source install/setup.bash

(三) 话题通信

  • ros2内常使用话题进行通信，以发布——订阅的方式进行通信

根据发布者（Publisher）和订阅者（Subscriber）的数量关系，主要有以下4种通信模型：

1. 一对一通信 (1-to-1)

最基础的通信方式，1个发布者通过话题发数据，1个订阅者接收。

发布者 (Node)
→
话题 (Topic)
→
订阅者 (Node)

2. 一对多通信 (1-to-Many)

1个发布者发送数据，多个订阅者同时接收，例如：单个激光雷达节点发布点云，导航和避障节点同时订阅。

发布者
→
话题
→
订阅者 A
订阅者 B

3. 多对一通信 (Many-to-1)

多个发布者向同一个话题发数据，1个订阅者统一接收，例如：不同方位的传感器把状态发到同一个报警话题，总控制台统一监控。

发布者 A
发布者 B
→
话题
→
订阅者

4. 多对多通信 (Many-to-Many)

多个发布者向同一个话题发数据，同时也有多个订阅者接收，类似总线结构。

发布者 A
发布者 B
→
话题
→
订阅者 A
订阅者 B
当然ros2也支持自身订阅自身。

#include "rclcpp/rclcpp.hpp"
#include "std_msgs/msg/string.hpp"

class ExamplePublisher : public rclcpp::Node {
   public:
    ExamplePublisher(std::string name) : Node(name) {
        RCLCPP_INFO(this->get_logger(), "%s has been started.", name.c_str());

        publisher_ = this->create_publisher<std_msgs::msg::String>("pub", 10);
        timer_ = this->create_wall_timer(std::chrono::milliseconds(500), [this]() { timer_callback(); });
    }

   private:
    rclcpp::Publisher<std_msgs::msg::String>::SharedPtr publisher_;
    rclcpp::TimerBase::SharedPtr timer_;
    void timer_callback() {
        auto message = std_msgs::msg::String();
        message.data = "Hello, ROS2!";
        RCLCPP_INFO(this->get_logger(), "Publishing: '%s'", message.data.c_str());
        publisher_->publish(message);
    }
};

int main(int argc, char** argv) {
    rclcpp::init(argc, argv);
    auto node = std::make_shared<ExamplePublisher>("publisher");
    rclcpp::spin(node);
    rclcpp::shutdown();
}

import rclpy
from rclpy.node import Node
from std_msgs.msg import String


class ExamplePublisher(Node):
    def __init__(self):
        super().__init__("py_publisher")
        self.publisher_ = self.create_publisher(String, "chatter", 10)
        self.timer_ = self.create_timer(0.5, self.timer_callback)
        self.count_ = 0

    def timer_callback(self):
        msg = String()
        msg.data = f"Hello ROS2 ({self.count_})"
        self.publisher_.publish(msg)
        self.get_logger().info(f"Publishing: {msg.data}")
        self.count_ += 1


def main(args=None):
    rclpy.init(args=args)
    node = ExamplePublisher()
    rclpy.spin(node)
    node.destroy_node()
    rclpy.shutdown()


if __name__ == "__main__":
    main()

(四) 服务通信 (Service)

  • service 更像“打电话问问题”：请求（request）发过去，对方处理完给你一个响应（response）
  • 适合：查询一次性的状态（比如“当前电量多少？”、“是否已经到位？”）

  • 查看service接口 
    1

    ros2 interface show <pkg/srv/xxx>
  • 调用service（手动发请求） 
    1

    ros2 service call <service_name> <service_type> "{field1: value1, field2: value2}"

#include <chrono>
#include <memory>

#include <rclcpp/rclcpp.hpp>
#include <example_interfaces/srv/add_two_ints.hpp>

class AddTwoIntsServer : public rclcpp::Node {
   public:
  AddTwoIntsServer() : Node("add_two_ints_server_cpp") {
    service_ = this->create_service<example_interfaces::srv::AddTwoInts>(
      "add_two_ints",
      [this](
        const std::shared_ptr<example_interfaces::srv::AddTwoInts::Request> request,
        std::shared_ptr<example_interfaces::srv::AddTwoInts::Response> response) {
        response->sum = request->a + request->b;
        RCLCPP_INFO(
          this->get_logger(),
          "Request: a=%lld, b=%lld -> sum=%lld",
          (long long)request->a,
          (long long)request->b,
          (long long)response->sum);
      });

    RCLCPP_INFO(this->get_logger(), "AddTwoInts service ready.");
  }

   private:
  rclcpp::Service<example_interfaces::srv::AddTwoInts>::SharedPtr service_;
};

int main(int argc, char** argv) {
  rclcpp::init(argc, argv);
  auto node = std::make_shared<AddTwoIntsServer>();
  rclcpp::spin(node);
  rclcpp::shutdown();
  return 0;
}

#include <chrono>
#include <memory>

#include <rclcpp/rclcpp.hpp>
#include <example_interfaces/srv/add_two_ints.hpp>

int main(int argc, char** argv) {
  using namespace std::chrono_literals;

  rclcpp::init(argc, argv);
  auto node = rclcpp::Node::make_shared("add_two_ints_client_cpp");

  auto client = node->create_client<example_interfaces::srv::AddTwoInts>("add_two_ints");
  while (!client->wait_for_service(1s)) {
    if (!rclcpp::ok()) {
      RCLCPP_ERROR(node->get_logger(), "Interrupted while waiting for the service.");
      rclcpp::shutdown();
      return 1;
    }
    RCLCPP_INFO(node->get_logger(), "Waiting for service...");
  }

  auto request = std::make_shared<example_interfaces::srv::AddTwoInts::Request>();
  request->a = 12;
  request->b = 30;

  auto future = client->async_send_request(request);
  auto ret = rclcpp::spin_until_future_complete(node, future);
  if (ret == rclcpp::FutureReturnCode::SUCCESS) {
    auto response = future.get();
    RCLCPP_INFO(node->get_logger(), "Result: %lld", (long long)response->sum);
  } else {
    RCLCPP_ERROR(node->get_logger(), "Service call failed.");
  }

  rclcpp::shutdown();
  return 0;
}

import rclpy
from rclpy.node import Node
from example_interfaces.srv import AddTwoInts


class AddTwoIntsClient(Node):
    def __init__(self):
        super().__init__("add_two_ints_client")
        self.client_ = self.create_client(AddTwoInts, "add_two_ints")
        while not self.client_.wait_for_service(timeout_sec=1.0):
            self.get_logger().info("Waiting for service...")

    def call(self, a: int, b: int):
        req = AddTwoInts.Request()
        req.a = a
        req.b = b
        future = self.client_.call_async(req)
        rclpy.spin_until_future_complete(self, future)
        return future.result()


def main(args=None):
    rclpy.init(args=args)
    node = AddTwoIntsClient()
    resp = node.call(12, 30)
    node.get_logger().info(f"Result: {resp.sum}")
    node.destroy_node()
    rclpy.shutdown()


if __name__ == "__main__":
    main()

(五) 动作通信 (Action)

  • action 更像“下单跑腿”：你发一个目标（Goal），它会边做边给反馈（Feedback），最后给结果（Result）
  • 适合：耗时动作（导航、机械臂运动、抓取）

1. action常用CLI工具

  • 查看action列表 
    1

    ros2 action list
  • 查看action类型 
    1

    ros2 action info <action_name>
  • 发送action目标 
    1

    ros2 action send_goal <action_name> <action_type> "{field: value}" --feedback

#include <chrono>
#include <functional>
#include <memory>
#include <thread>
#include <vector>

#include <rclcpp/rclcpp.hpp>
#include <rclcpp_action/rclcpp_action.hpp>

#include <example_interfaces/action/fibonacci.hpp>

class FibonacciActionServer : public rclcpp::Node {
   public:
  using Fibonacci = example_interfaces::action::Fibonacci;
  using GoalHandleFibonacci = rclcpp_action::ServerGoalHandle<Fibonacci>;

  FibonacciActionServer() : Node("fibonacci_action_server_cpp") {
    using std::placeholders::_1;
    using std::placeholders::_2;

    server_ = rclcpp_action::create_server<Fibonacci>(
      this,
      "fibonacci",
      std::bind(&FibonacciActionServer::handle_goal, this, _1, _2),
      std::bind(&FibonacciActionServer::handle_cancel, this, _1),
      std::bind(&FibonacciActionServer::handle_accepted, this, _1));

    RCLCPP_INFO(this->get_logger(), "Fibonacci action server ready.");
  }

   private:
  rclcpp_action::Server<Fibonacci>::SharedPtr server_;

  rclcpp_action::GoalResponse handle_goal(
    const rclcpp_action::GoalUUID& /*uuid*/,
    std::shared_ptr<const Fibonacci::Goal> goal) {
    RCLCPP_INFO(this->get_logger(), "Received goal, order=%d", goal->order);
    if (goal->order < 0) {
      return rclcpp_action::GoalResponse::REJECT;
    }
    return rclcpp_action::GoalResponse::ACCEPT_AND_EXECUTE;
  }

  rclcpp_action::CancelResponse handle_cancel(const std::shared_ptr<GoalHandleFibonacci> /*goal_handle*/) {
    RCLCPP_INFO(this->get_logger(), "Received cancel request");
    return rclcpp_action::CancelResponse::ACCEPT;
  }

  void handle_accepted(const std::shared_ptr<GoalHandleFibonacci> goal_handle) {
    std::thread{[this, goal_handle]() { execute(goal_handle); }}.detach();
  }

  void execute(const std::shared_ptr<GoalHandleFibonacci> goal_handle) {
    using namespace std::chrono_literals;

    const auto goal = goal_handle->get_goal();
    auto feedback = std::make_shared<Fibonacci::Feedback>();
    auto result = std::make_shared<Fibonacci::Result>();

    std::vector<int32_t> seq;
    seq.push_back(0);
    if (goal->order > 0) {
      seq.push_back(1);
    }

    rclcpp::Rate rate(2.0);
    for (int i = 2; i <= goal->order && rclcpp::ok(); ++i) {
      if (goal_handle->is_canceling()) {
        result->sequence = seq;
        goal_handle->canceled(result);
        RCLCPP_INFO(this->get_logger(), "Goal canceled");
        return;
      }

      seq.push_back(seq[i - 1] + seq[i - 2]);
      feedback->sequence = seq;
      goal_handle->publish_feedback(feedback);
      RCLCPP_INFO(this->get_logger(), "Feedback size=%zu", seq.size());
      rate.sleep();
    }

    result->sequence = seq;
    goal_handle->succeed(result);
    RCLCPP_INFO(this->get_logger(), "Goal succeeded");
  }
};

int main(int argc, char** argv) {
  rclcpp::init(argc, argv);
  rclcpp::spin(std::make_shared<FibonacciActionServer>());
  rclcpp::shutdown();
  return 0;
}

int main(int argc, char** argv) {
  using namespace std::chrono_literals;

  rclcpp::init(argc, argv);
  auto node = rclcpp::Node::make_shared("fibonacci_action_client_cpp");

  using Fibonacci = example_interfaces::action::Fibonacci;
  using GoalHandleFibonacci = rclcpp_action::ClientGoalHandle<Fibonacci>;

  auto client = rclcpp_action::create_client<Fibonacci>(node, "fibonacci");
  if (!client->wait_for_action_server(5s)) {
    RCLCPP_ERROR(node->get_logger(), "Action server not available.");
    rclcpp::shutdown();
    return 1;
  }

  Fibonacci::Goal goal;
  goal.order = 10;

  rclcpp_action::Client<Fibonacci>::SendGoalOptions options;
  options.feedback_callback =
    [node](GoalHandleFibonacci::SharedPtr /*handle*/, const std::shared_ptr<const Fibonacci::Feedback> feedback) {
      RCLCPP_INFO(node->get_logger(), "Feedback len=%zu", feedback->sequence.size());
    };
  options.result_callback =
    [node](const GoalHandleFibonacci::WrappedResult& result) {
      if (result.code == rclcpp_action::ResultCode::SUCCEEDED) {
        RCLCPP_INFO(node->get_logger(), "Result len=%zu", result.result->sequence.size());
      } else {
        RCLCPP_WARN(node->get_logger(), "Result code=%d", (int)result.code);
      }
      rclcpp::shutdown();
    };

  client->async_send_goal(goal, options);
  rclcpp::spin(node);
  return 0;
}

import time

import rclpy
from rclpy.node import Node
from rclpy.action import ActionServer

from example_interfaces.action import Fibonacci


class FibonacciActionServer(Node):
  def __init__(self):
    super().__init__("fibonacci_action_server_py")
    self.server_ = ActionServer(self, Fibonacci, "fibonacci", self.execute_callback)
    self.get_logger().info("Fibonacci action server ready.")

  def execute_callback(self, goal_handle):
    order = goal_handle.request.order
    self.get_logger().info(f"Received goal, order={order}")

    feedback_msg = Fibonacci.Feedback()
    seq = [0]
    if order > 0:
      seq.append(1)

    for i in range(2, order + 1):
      if goal_handle.is_cancel_requested:
        goal_handle.canceled()
        result = Fibonacci.Result()
        result.sequence = seq
        self.get_logger().info("Goal canceled")
        return result

      seq.append(seq[i - 1] + seq[i - 2])
      feedback_msg.sequence = seq
      goal_handle.publish_feedback(feedback_msg)
      time.sleep(0.5)

    goal_handle.succeed()
    result = Fibonacci.Result()
    result.sequence = seq
    self.get_logger().info("Goal succeeded")
    return result


def main(args=None):
  rclpy.init(args=args)
  node = FibonacciActionServer()
  rclpy.spin(node)
  node.destroy_node()
  rclpy.shutdown()


if __name__ == "__main__":
  main()

import rclpy
from rclpy.node import Node
from rclpy.action import ActionClient

from example_interfaces.action import Fibonacci


class FibonacciActionClient(Node):
  def __init__(self):
    super().__init__("fibonacci_action_client_py")
    self.client_ = ActionClient(self, Fibonacci, "fibonacci")

  def send_goal(self, order: int):
    goal = Fibonacci.Goal()
    goal.order = order

    self.client_.wait_for_server()

    self.get_logger().info(f"Sending goal, order={order}")
    return self.client_.send_goal_async(goal, feedback_callback=self.feedback_callback)

  def feedback_callback(self, feedback_msg):
    seq = feedback_msg.feedback.sequence
    self.get_logger().info(f"Feedback len={len(seq)}")


def main(args=None):
  rclpy.init(args=args)
  node = FibonacciActionClient()

  future = node.send_goal(10)
  rclpy.spin_until_future_complete(node, future)

  goal_handle = future.result()
  if not goal_handle.accepted:
    node.get_logger().error("Goal rejected")
    node.destroy_node()
    rclpy.shutdown()
    return

  result_future = goal_handle.get_result_async()
  rclpy.spin_until_future_complete(node, result_future)
  result = result_future.result().result
  node.get_logger().info(f"Result len={len(result.sequence)}")

  node.destroy_node()
  rclpy.shutdown()


if __name__ == "__main__":
  main()

(七) Launch 文件（启动多个节点）

当你需要同时启动很多节点，并且要配置参数、重映射、命名空间、引入别的launch时，手敲 ros2 run ... 就会非常痛苦。 所以：launch 就是“启动脚本”，统一管理一套启动方式。

ROS2 常见的 launch 有 2 类：

  • XML Launch：xxx.launch.xml

    ros2 pkg prefix <package_name>

    prefix 告诉你 install 后的位置，launch 文件一般会在 share/<package_name>/launch/ 下。

from launch import LaunchDescription
from launch.actions import DeclareLaunchArgument
from launch.substitutions import LaunchConfiguration
from launch_ros.actions import Node


def generate_launch_description():
    ns_arg = DeclareLaunchArgument("ns", default_value="robot1")
    rate_arg = DeclareLaunchArgument("rate", default_value="2.0")

    node = Node(
        package="<package_name>",
        executable="param_node",
        namespace=LaunchConfiguration("ns"),
        parameters=[{"rate": LaunchConfiguration("rate")}],
        output="screen",
    )

    return LaunchDescription([ns_arg, rate_arg, node])

from ament_index_python.packages import get_package_share_directory
import os


def generate_launch_description():
    pkg_share = get_package_share_directory("<package_name>")
